import numpy as np

# 定义权函数 rho(x) = x
def inner_product(g, h, a=0, b=1):
    """计算加权内积 (g, h) = ∫_0^1 g(x)h(x) * x dx"""
    # 使用数值积分，这里使用高斯积分保证精度
    x, w = np.polynomial.legendre.leggauss(100)
    # 将区间从[-1,1]映射到[0,1]
    t = 0.5 * (x + 1)
    dt_dx = 0.5
    # 计算被积函数
    integrand = g(t) * h(t) * t * dt_dx * 2  # 乘以2是因为权重归一化
    return np.sum(integrand * w)

# 定义基函数
def phi0(x):
    return np.ones_like(x)

def phi1(x):
    return x

# 定义目标函数
def f(x):
    return x**2

# 计算法方程的矩阵元素
A = np.zeros((2, 2))
b = np.zeros(2)

# 计算内积
A[0, 0] = inner_product(phi0, phi0)
A[0, 1] = inner_product(phi0, phi1)
A[1, 0] = A[0, 1]  # 对称性
A[1, 1] = inner_product(phi1, phi1)

b[0] = inner_product(f, phi0)
b[1] = inner_product(f, phi1)

print("法方程矩阵 A:")
print(A)
print("\n右端向量 b:")
print(b)

# 求解线性方程组
coeff = np.linalg.solve(A, b)
print(f"\n求解系数: a0 = {coeff[0]:.6f}, a1 = {coeff[1]:.6f}")

# 与我的解析解对比
my_a0 = -0.6  # 我最初的计算结果
my_a1 = 1.2

print(f"\n我的解析解: a0 = {my_a0}, a1 = {my_a1}")
print(f"数值解: a0 = {coeff[0]:.6f}, a1 = {coeff[1]:.6f}")

# 计算相对误差
rel_error_a0 = abs(coeff[0] - my_a0) / abs(coeff[0])
rel_error_a1 = abs(coeff[1] - my_a1) / abs(coeff[1])
print(f"\n相对误差: a0: {rel_error_a0:.2%}, a1: {rel_error_a1:.2%}")

# 验证法方程解析形式
print("\n=== 解析计算验证 ===")
# 解析计算内积
# (1,1) = ∫_0^1 1*1*x dx = 1/2
# (1,x) = ∫_0^1 1*x*x dx = ∫_0^1 x^2 dx = 1/3
# (x,x) = ∫_0^1 x*x*x dx = ∫_0^1 x^3 dx = 1/4
# (f,1) = ∫_0^1 x^2*1*x dx = ∫_0^1 x^3 dx = 1/4
# (f,x) = ∫_0^1 x^2*x*x dx = ∫_0^1 x^4 dx = 1/5

A_analytic = np.array([[1/2, 1/3], [1/3, 1/4]])
b_analytic = np.array([1/4, 1/5])
coeff_analytic = np.linalg.solve(A_analytic, b_analytic)
print(f"解析法方程系数: a0 = {coeff_analytic[0]:.6f}, a1 = {coeff_analytic[1]:.6f}")

# 直接解方程
print("\n=== 手动解方程验证 ===")
# 方程:
# 0.5*a0 + 1/3*a1 = 0.25   (1)
# 1/3*a0 + 0.25*a1 = 0.2    (2)
# 从(1)得：a1 = (0.25 - 0.5*a0) * 3 = 0.75 - 1.5*a0
# 代入(2)：1/3*a0 + 0.25*(0.75 - 1.5*a0) = 0.2
# 1/3*a0 + 0.1875 - 0.375*a0 = 0.2
# (1/3 - 0.375)a0 = 0.2 - 0.1875
# (0.333333 - 0.375)a0 = 0.0125
# -0.041667*a0 = 0.0125
# a0 = -0.3
# a1 = 0.75 - 1.5*(-0.3) = 0.75 + 0.45 = 1.2
print("正确解应为: a0 = -0.3, a1 = 1.2")
